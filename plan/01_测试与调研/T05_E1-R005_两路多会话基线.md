```bash
MS="$PY $SCR/multisession_probe.py"
T2=/data2/_shared/runs/ft_rounds/E1-R003-v2/trace.json
FLAGS="--prefill-layer-major --max-prefill-length 4096 --enable-special-token-ckpt"
suite() { local ARM=$1; local O=$D/$ARM
  ok || exit 7
  start $O "FT_DEV_D0=1 FT_DEV_D0_PATH=$O/d0 FT_DEV_D0_TIMELINE=1 FT_DEV_D0_TIMELINE_STEPS=20000 FT_DEV_PAUSE_ENDPOINT=1" $FLAGS
  [ $READY = 1 ] || { echo "$ARM NOT-READY"; stop; exit 6; }

  $PY $SCR/probe_ft_cold32k.py "$RID-$ARM-first" 32052 128 nothink > $O/first_shot.log 2>&1; echo "$ARM first rc=$?"; tail -3 $O/first_shot.log
  $MS two-decode --ctx 32000 --max-tokens 512 --same-prompt 0 --reps 3 --out $O/a_two_decode_diff.json > $O/a.log 2>&1; echo "$ARM a-diff rc=$?"
  $MS two-decode --ctx 32000 --max-tokens 512 --same-prompt 1 --reps 1 --out $O/a_two_decode_same.json >> $O/a.log 2>&1; echo "$ARM a-same rc=$?"
  $MS stall --decode-ctx 4000 --decode-max-tokens 1500 --prefill-kind cold32k --prefill-target 32052 --delay 3 --reps 3 --out $O/b_stall_cold32k.json > $O/b.log 2>&1; echo "$ARM b-cold rc=$?"
  $MS stall --decode-ctx 4000 --decode-max-tokens 1500 --prefill-kind agent_turn --trace $T2 --round 6 --delay 3 --reps 3 --out $O/b_stall_agent.json >> $O/b.log 2>&1; echo "$ARM b-agent rc=$?"
  local P0=$(grep -c "Prefill batch" $O/serve.log)
  $MS pair-short --trace $T2 --rounds 3,4 --reps 3 --out $O/c_pair_short.json > $O/c.log 2>&1; echo "$ARM c rc=$?"
  local P1=$(grep -c "Prefill batch" $O/serve.log)
  grep "Prefill batch" $O/serve.log | sed -n "$((P0+1)),${P1}p" > $O/c_prefill_lines.txt
  $MS h1 --H 1,8,32 --reps 3 --prefill-target 32052 --out $O/d_h1.json > $O/d.log 2>&1; echo "$ARM d rc=$?"
  for f in a_two_decode_diff b_stall_cold32k b_stall_agent c_pair_short d_h1; do [ -s $O/$f.json ] || { echo "$ARM GATE-FAIL-MISSING $f"; stop; exit 5; }; done
  stop
}
suite P1
suite P2
$PY $SCR/serve_args_diff.py $D/P1/serve.log $D/P2/serve.log > $D/args_diff_P1P2.txt 2>&1; cat $D/args_diff_P1P2.txt
```
