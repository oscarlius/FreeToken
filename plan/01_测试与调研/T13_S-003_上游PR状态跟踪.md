```bash
for n in 577 578 505; do
  curl -s https://api.github.com/repos/FlashML-org/FreeToken/pulls/$n | python -c "import json,sys;d=json.load(sys.stdin);print($n, d.get('state'), 'merged=',d.get('merged'), 'merged_at=',d.get('merged_at'), 'head=',d.get('head',{}).get('sha','')[:7], 'updated=',d.get('updated_at'), 'title=',d.get('title'))"
  curl -s "https://api.github.com/repos/FlashML-org/FreeToken/pulls/$n/commits?per_page=100" | python -c "import json,sys;print('  commits:', [c['sha'][:7] for c in json.load(sys.stdin)])"
  curl -s "https://api.github.com/repos/FlashML-org/FreeToken/issues/$n/comments?per_page=100" | python -c "import json,sys;L=json.load(sys.stdin);print('  comments:', len(L), 'last:', (L[-1]['created_at'], L[-1]['user']['login']) if L else None)"
done
curl -s "https://api.github.com/repos/FlashML-org/FreeToken/commits?per_page=5" | python -c "import json,sys;[print('main', c['sha'][:7], c['commit']['author']['date'], c['commit']['message'].splitlines()[0][:80]) for c in json.load(sys.stdin)]"
```
